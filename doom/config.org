** Core
The command =save-buffer= is bound to =C-x C-s=, while =save-some-buffers= is
bound to =C-x s=. These keybindings are easy to mix up while typing; that is
annoying since the latter command steals focus to the minibuffer while waiting
for a response.

#+begin_src emacs-lisp :tangle yes
(global-unset-key (kbd "C-x s"))
#+end_src

A buffer from =compile= with COMINT shares =comint-mode-map= with every comint
buffer, so =local-set-key= there leaks into all of them. =doom/reload= does this
with =q= when it finishes, which made =q= quit agent-shell, =shell= and REPL
buffers, even in insert state. Giving each such buffer its own keymap keeps the
binding local.

#+begin_src emacs-lisp :tangle yes
(add-hook 'compilation-shell-minor-mode-hook
          (defun my/compilation-private-keymap-h ()
            (use-local-map (make-composed-keymap nil (current-local-map)))))

#+begin_src emacs-lisp :tangle yes
(after! flymake
  (defun custom-flymake-popon-format-diagnostic (diagnostic)
    "Format DIAGNOSTIC to text."
    (with-temp-buffer
      (insert
       "* "
       (propertize
        (string-trim (string-join (split-string (flymake-diagnostic-text diagnostic '(message)) "[\n\r]") ": "))
        'face (flymake--lookup-type-property
               (flymake-diagnostic-type diagnostic) 'mode-line-face)))

      ;; Break long lines.
      (goto-char (point-min))
      (let ((last-whitespace nil)
            (last-char-whitespace-p nil))
        (while (not (eobp))
          (if (not (member (buffer-substring-no-properties
                            (point) (1+ (point)))
                           '(" " "\t")))
              (setq last-char-whitespace-p nil)
            (unless last-char-whitespace-p
              (setq last-whitespace (point))
              (setq last-char-whitespace-p t)))
          (when (>= (current-column) flymake-popon-width)
            (save-excursion
              (when last-whitespace
                (goto-char last-whitespace))
              (insert "\n  ")
              (while (member (buffer-substring-no-properties
                              (point) (1+ (point)))
                             '(" " "\t"))
                (delete-char 1))))
          (right-char)))
      (buffer-string))))

(setq flymake-popon-diagnostic-formatter #'custom-flymake-popon-format-diagnostic)


#+end_src

** Git
magit's "c g" (generate commit) uses gptel-magit, which defaults to a
conventional-commits prompt that allows types (e.g. build, ci, perf) and a
freeform body that the global commit-msg hook
(=~/.config/git/hooks/commit-msg=) rejects. Build the prompt from
=~/.claude/skills/commit/FORMAT.md= instead — the same file the Claude Code
=commit= skill reads for its message shape — so the two never drift apart.

#+begin_src emacs-lisp :tangle yes
(defun my/gptel-magit-commit-prompt ()
  "Build the gptel-magit commit prompt from the shared commit format spec.
Reads ~/.claude/skills/commit/FORMAT.md, the same file the Claude Code
`commit' skill uses, so magit's \"c g\" and Claude's /commit produce the
same commit message shape."
  (let ((format-spec
         (with-temp-buffer
           (insert-file-contents
            (expand-file-name "~/.claude/skills/commit/FORMAT.md"))
           (string-trim (buffer-string)))))
    (concat
     "You are an expert at writing Git commits. Your job is to write a short clear commit message that summarizes the changes.\n\n"
     "Only return the commit message in your response. Do not include any additional meta-commentary about the task. Do not include the raw diff output in the commit message.\n\n"
     "Follow this format exactly:\n\n"
     format-spec
     "\n\nReturn the raw commit message as plain text. Do not wrap it in a code block or backticks; the fences in the format above only delimit the examples.")))

(use-package! gptel-magit
  :hook (magit-mode . gptel-magit-install)
  :config
  (setq gptel-magit-commit-prompt (my/gptel-magit-commit-prompt)))

(defun my/git-worktree-list (dir)
  "Return the worktree paths of the git repo containing DIR.
Return nil when DIR is not inside a git repo (or git is unavailable)."
  (let ((default-directory dir))
    (when (executable-find "git")
      (ignore-errors
        (delq nil
              (mapcar
               (lambda (line)
                 (when (string-prefix-p "worktree " line)
                   (substring line (length "worktree "))))
               (process-lines "git" "worktree" "list" "--porcelain")))))))

(defvar my/projectile-resolve-worktree-p t
  "When non-nil, switching projectile project asks which git worktree to
open if the target repository has more than one. Let-bound to nil by
`my/projectile-switch-git-worktree', which already asks explicitly.")

(defun my/projectile-resolve-worktree (project-to-switch)
  "Resolve PROJECT-TO-SWITCH to one of its git worktrees.
Prompt when the project's repository has more than one worktree;
return PROJECT-TO-SWITCH unchanged otherwise."
  (let ((worktrees (and my/projectile-resolve-worktree-p
                        (my/git-worktree-list project-to-switch))))
    (if (> (length worktrees) 1)
        (completing-read "Worktree: " worktrees nil t)
      project-to-switch)))

(advice-add 'projectile-switch-project-by-name :filter-args
            (lambda (args)
              (cons (my/projectile-resolve-worktree (car args)) (cdr args))))

(defun my/projectile-switch-git-worktree ()
  "Switch Projectile directly to a Git worktree of the current repository."
  (interactive)
  (let* ((default-directory (projectile-project-root))
         (worktrees (my/git-worktree-list default-directory))
         (choice (completing-read "Worktree: " worktrees nil t)))
    (let ((my/projectile-resolve-worktree-p nil))
      (projectile-switch-project-by-name choice))))

(map! :leader
      :desc "Worktrees"
      "p w" #'my/projectile-switch-git-worktree)

=W= on an entry opens an agent-shell session in the org repo with a prompt to
summarize the entry's own link (not elfeed's internal id) via the
=roam-content-summarizer= Claude Code subagent. The prompt is inserted but not
submitted, so it can be reviewed first; submit with =agent-shell-submit=
(=C-return=).
#+begin_src emacs-lisp :tangle yes
(defvar my/elfeed-wiki-repo (expand-file-name "~/documents/org/")
  "Org repo holding the roam wiki and the `roam-content-summarizer' agent.")

(defun my/elfeed-summarize-entry-to-wiki ()
  "Open an agent-shell session to summarize the current entry into the wiki.
Starts a new agent-shell rooted at `my/elfeed-wiki-repo' with a prompt asking
the `roam-content-summarizer' subagent to summarize the entry's own link."
  (interactive)
  (require 'agent-shell)
  (let ((link (elfeed-entry-link elfeed-show-entry)))
    (unless link
      (user-error "No link on this entry"))
    (let* ((default-directory my/elfeed-wiki-repo)
           (shell-buffer (agent-shell--new-shell :location my/elfeed-wiki-repo)))
      (agent-shell--display-and-insert-context
       shell-buffer
       (format "Use the roam-content-summarizer subagent to summarize %s and add it to the wiki."
               link)))))

(after! elfeed
  (map! :map elfeed-show-mode-map
        :n "W" #'my/elfeed-summarize-entry-to-wiki))
